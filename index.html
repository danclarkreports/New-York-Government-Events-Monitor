Python Automation Script

Production-ready script. Uses google-genai to parse data, handles rate limits, and exports JSON.

import os
import time
import json
import requests
from bs4 import BeautifulSoup
from google import genai
from google.genai import types

GEMINI_API_KEY = os.getenv("GEMINI_API_KEY")
client = genai.Client(api_key=GEMINI_API_KEY)

def load_sources():
    with open("sources.json", "r") as f:
        return json.load(f).get("sources", [])

def extract_events_with_gemini(html_content, org_name):
    prompt = f"Extract all future public hearings, board meetings, and council events from the following HTML for {org_name}."
    response = client.models.generate_content(
        model='gemini-2.5-pro',
        contents=[prompt, html_content[:30000]],
        config=types.GenerateContentConfig(response_mime_type="application/json", temperature=0.1)
    )
    return json.loads(response.text)

if __name__ == "__main__":
    print("Running scraper workflow...")
